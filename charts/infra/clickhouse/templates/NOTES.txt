Quick Test:
==============

# Port-forward HTTP interface
kubectl port-forward svc/{{ $fullname }} 8123:{{ .Values.service.httpPort }} -n {{ .Release.Namespace }}

# Test connection
curl "http://localhost:8123/ping"

# Query via HTTP
curl "http://localhost:8123/?query=SELECT%201"

# With authentication
export CLICKHOUSE_PASSWORD=$(kubectl get secret {{ include "clickhouse.credentialsSecretName" . }} -n {{ .Release.Namespace }} -o jsonpath='{.data.CLICKHOUSE_PASSWORD}' | base64 -d)
curl "http://{{ .Values.clickhouse.defaultUser }}:${CLICKHOUSE_PASSWORD}@localhost:8123/?query=SHOW%20DATABASES"
